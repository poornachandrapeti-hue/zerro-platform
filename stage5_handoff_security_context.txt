 797: def reconcile_approved_quality_case(
 798:     pid: int,
 799:     receipt_line_id: int,
 800:     expected_version: int = 1,
 801:     u: User = Depends(authenticated),
 802:     db: Session = Depends(get_db),
 803: ):
 804:     """One-time, explicit QA handoff. Never releases, posts, or issues inventory.
 805: 
 806:     Fail closed for revoked/previously decided cases. Subsequent Quality changes
 807:     require a separately reviewed reinspection workflow, not silent overwrite.
 808:     """
 809:     project = qa_access(db, u, pid, edit=True)
 810:     from app.domains.quality.material import _totals
 811:     from app.domains.quality.models import MaterialInspection
 812: 
 813:     line = (db.query(GoodsReceiptLine)
 814:         .join(GoodsReceipt, GoodsReceipt.id == GoodsReceiptLine.goods_receipt_id)
 815:         .filter(GoodsReceiptLine.id == receipt_line_id,
 816:                 GoodsReceipt.project_id == pid,
 817:                 GoodsReceipt.organization_id == u.organization_id,
 818:                 GoodsReceipt.status == "Posted")
 819:         .with_for_update().first())
 820:     if line is None:
 821:         raise HTTPException(404, "Posted goods receipt line not found")
 822:     case = (db.query(StoresInspectionCase)
 823:         .filter_by(goods_receipt_line_id=receipt_line_id,
 824:                    project_id=pid, organization_id=u.organization_id)
 825:         .with_for_update().first())
 826:     if case and case.status == "Quality Revoked":
 827:         raise HTTPException(409, "Revoked QA case requires separately authorized reinspection")
 828:     if case and case.version != expected_version:
 829:         raise HTTPException(409, "Stores QA case version changed")
 830:     if case and db.query(StoresInspectionDecision.id).filter_by(case_id=case.id).first():
 831:         raise HTTPException(409, "Existing Stores decisions require manual reconciliation")
 832:     if case and any(Decimal(str(getattr(case, key))) != value for key, value in (
 833:         ("pending_qty", Decimal(str(line.quantity))),
 834:         ("accepted_qty", Decimal(0)),
 835:         ("rejected_qty", Decimal(0)),
 836:         ("quarantined_qty", Decimal(0)),
 837:     )):
 838:         raise HTTPException(409, "Stores case already contains dispositions")
 839:     if db.query(MaterialInspection.id).filter(
 840:         MaterialInspection.goods_receipt_id == line.goods_receipt_id,
 841:         MaterialInspection.status.in_(("Draft", "Pending Review", "Returned for Correction"))
 842:     ).first():

 845:     received = Decimal(str(line.quantity))
 846:     approved = amounts["accepted"] + amounts["rejected"] + amounts["quarantined"]
 847:     if approved <= 0 or approved > received or any(q < 0 for q in amounts.values()):
 848:         raise HTTPException(409, "No valid approved Quality disposition to reconcile")
 849:     if case is None:

 851:             organization_id=u.organization_id, project_id=pid,
 852:             goods_receipt_line_id=receipt_line_id, pending_qty=received,

 860:     case.quarantined_qty = amounts["quarantined"]
 861:     case.status = "Pending Inspection" if case.pending_qty > 0 else "Disposition Recorded"
 862:     case.version += 1
