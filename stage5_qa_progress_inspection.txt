  27:                                authorized, server_revalidated):
  28:     """Return read-only eligibility assessment; never authorize billing or payment."""
  29:     if authorized is not True or server_revalidated is not True:
  30:         raise CommercialBoundaryError("Server authorization and fresh source validation required")
  31:     if not isinstance(progress, dict) or not isinstance(contract, dict):
  32:         raise CommercialBoundaryError("Canonical progress and contract records required")
  33:     for record in (progress, contract):
  34:         if (record.get("organization_id"), record.get("project_id")) != (organization_id, project_id):
  35:             raise CommercialBoundaryError("Cross-scope authority")
  36:         if record.get("server_authorized") is not True:
  37:             raise CommercialBoundaryError("Missing server-grounded source")
  38:     if progress.get("status") != "Published" or progress.get("published") is not True:
  39:         raise CommercialBoundaryError("Only persisted Published snapshots may enter commercial review")
  40:     if progress.get("publication_persisted") is not True or progress.get("approval_audit_valid") is not True:
  41:         raise CommercialBoundaryError("Published snapshot lacks persisted audit authority")
  42:     if progress.get("source_integrity_valid") is not True or progress.get("revoked") is True:
  43:         raise CommercialBoundaryError("Measurement evidence invalid or revoked")
  44:     if contract.get("status") != "Active" or contract.get("billing_terms_approved") is not True:
  45:         raise CommercialBoundaryError("Contract is not approved for billing review")
  46:     if not progress.get("snapshot_id") or not progress.get("revision_id") or not contract.get("contract_id"):
  47:         raise CommercialBoundaryError("Missing immutable source identifiers")
  48:     if progress.get("contract_id") != contract.get("contract_id"):
  49:         raise CommercialBoundaryError("Contract reference mismatch")
  50:     if progress.get("work_package_id") != contract.get("work_package_id"):
  51:         raise CommercialBoundaryError("Work package mismatch")

  54:         raise CommercialBoundaryError("Quantity unit mismatch")
